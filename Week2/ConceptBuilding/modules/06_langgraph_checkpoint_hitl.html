Setup & mental model

LangGraph turns an agent loop into an explicit graph: nodes mutate a shared TypedDict state; edges choose the next node. A checkpointer stores snapshots keyed by thread_id so you can pause (HITL) and resume later. 

# ASCII map — underwriting-style flow (Day 2 AM pattern)
#
#   START
#     |
#     v
#  [intake] ----> [risk_check] ----+----> [auto_approve] -------> END
#                                  |
#                                  +----> [request_documents] --> END
#                                  |
#                                  +----> [human_approve] ------> END
#                                            ^
#                                            |
#                         interrupt_before=["human_approve"]
#                         resume: update_state(...); invoke(None, config)

Lesson B2 — Fixed edges vs conditional edges

# Fixed:      add_edge("a", "b")
# Conditional: add_conditional_edges("gate", router_fn)
#              router_fn(state) returns the *name* of the next node
#
#   START --> [gate] --+--> [send]   --> END
#                      |
#                      +--> [revise] --> END

class HitlState(TypedDict):
    approved: bool | None
    path: str


def gate(state: HitlState) -> dict:
    return {}


def send(state: HitlState) -> dict:
    return {"path": "sent"}


def revise(state: HitlState) -> dict:
    return {"path": "revise"}


def should_send(state: HitlState) -> Literal["send", "revise"]:
    return "send" if state.get("approved") is True else "revise"


b = StateGraph(HitlState)
b.add_node("gate", gate)
b.add_node("send", send)
b.add_node("revise", revise)
b.add_edge(START, "gate")
b.add_conditional_edges(
    "gate",
    should_send,
    {"send": "send", "revise": "revise"},  # optional explicit map
)
b.add_edge("send", END)
b.add_edge("revise", END)
g = b.compile()

print("approved=True  ->", g.invoke({"approved": True, "path": ""}))
print("approved=None  ->", g.invoke({"approved": None, "path": ""}))

Lesson I1 — Full underwriting flow (no interrupt yet)

# ASCII — Day 2 AM underwriting_flow
#
# START -> intake -> risk_check -+-> auto_approve -------> END
#                               +-> request_documents --> END
#                               +-> human_approve ------> END


class FlowState(TypedDict):
    case_id: str
    case: dict
    recommended_action: str
    route: str
    event_trace: list[dict]


def intake(state: FlowState) -> dict:
    case = load_case(state["case_id"])
    return {
        "case": case,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "intake", "applicant": case["applicant"], "docs": case["documents_complete"]},
        ],
    }


def risk_check(state: FlowState) -> dict:
    action = state["case"]["recommended_action"]
    return {
        "recommended_action": action,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "risk_check", "risk_flags": state["case"]["risk_flags"], "recommended_action": action},
        ],
    }


def route_decision(
    state: FlowState,
) -> Literal["auto_approve", "request_documents", "human_approve"]:
    action = state["recommended_action"]
    if action == "auto_approve":
        return "auto_approve"
    if action == "request_documents":
        return "request_documents"
    return "human_approve"


def auto_approve(state: FlowState) -> dict:
    return {
        "route": "auto_approved",
        "event_trace": [*state.get("event_trace", []), {"node": "auto_approve", "route": "auto_approved"}],
    }


def request_documents(state: FlowState) -> dict:
    return {
        "route": "docs_requested",
        "event_trace": [*state.get("event_trace", []), {"node": "request_documents", "route": "docs_requested"}],
    }


def human_approve(state: FlowState) -> dict:
    return {
        "route": "pending_human_approval",
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "human_approve", "route": "pending_human_approval", "hitl": True},
        ],
    }


def build_underwriting():
    graph = StateGraph(FlowState)
    graph.add_node("intake", intake)
    graph.add_node("risk_check", risk_check)
    graph.add_node("auto_approve", auto_approve)
    graph.add_node("request_documents", request_documents)
    graph.add_node("human_approve", human_approve)
    graph.add_edge(START, "intake")
    graph.add_edge("intake", "risk_check")
    graph.add_conditional_edges("risk_check", route_decision)
    graph.add_edge("auto_approve", END)
    graph.add_edge("request_documents", END)
    graph.add_edge("human_approve", END)
    return graph.compile()

print(uw.invoke({
    "case_id": "UW-003",
    "case": {},
    "recommended_action": "",
    "route": "",
    "event_trace": [],
})["route"])

Lesson I2 — SqliteSaver + thread_id

Durable checkpoints need a SQLite connection (keep it open for the process lifetime) and a config with configurable.thread_id. Same thread id = same conversation / case resume handle. 

# ASCII — checkpoint identity
#
#  config = {"configurable": {"thread_id": "uw-hitl-demo"}}
#                 |
#                 v
#        +------------------+
#        | checkpoints.db   |  <-- SqliteSaver(connection)
#        |  thread snapshots|
#        +------------------+


class HitlUwState(TypedDict):
    case_id: str
    case: dict
    recommended_action: str
    approved: bool | None
    final_route: str
    event_trace: list[dict]


def hitl_intake(state: HitlUwState) -> dict:
    case = load_case(state["case_id"])
    return {
        "case": case,
        "recommended_action": case["recommended_action"],
        "event_trace": [*state.get("event_trace", []), {"node": "intake", "case_id": case["case_id"]}],
    }


def route_after_intake(state: HitlUwState) -> Literal["auto_finish", "human_approve"]:
    if state["recommended_action"] == "refer_underwriter":
        return "human_approve"
    return "auto_finish"


def auto_finish(state: HitlUwState) -> dict:
    route = state["recommended_action"]
    return {
        "final_route": route,
        "event_trace": [*state.get("event_trace", []), {"node": "auto_finish", "final_route": route}],
    }


def human_approve_node(state: HitlUwState) -> dict:
    approved = bool(state.get("approved"))
    route = "human_approved" if approved else "human_rejected"
    return {
        "final_route": route,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "human_approve", "approved": approved, "final_route": route},
        ],
    }


def build_hitl_graph(checkpointer: SqliteSaver):
    graph = StateGraph(HitlUwState)
    graph.add_node("intake", hitl_intake)
    graph.add_node("auto_finish", auto_finish)
    graph.add_node("human_approve", human_approve_node)
    graph.add_edge(START, "intake")
    graph.add_conditional_edges("intake", route_after_intake)
    graph.add_edge("auto_finish", END)
    graph.add_edge("human_approve", END)
    return graph.compile(
        checkpointer=checkpointer,
        interrupt_before=["human_approve"],
    )


Path("outputs").mkdir(exist_ok=True)

app = build_hitl_graph(saver)
config = {"configurable": {"thread_id": "uw-hitl-demo"}}

Lesson A1 — Multi-agent supervisor → intake → adjudication

Day 2 PM pattern on data/claim_narratives_10.json. Shared TypedDict state; intake owns claim_id / intake_queue; adjudication must not overwrite those fields. 

# ASCII — multi-agent claim flow
#
#   START
#     |
#     v
# [supervisor]  loads narrative text, assigns work
#     |
#     v
#   [intake]     sets claim_id + intake_queue  (owned fields)
#     |
#     v
# [adjudication] sets adjudication only; READ-ONLY on intake fields
#     |
#     v
#    END


def load_narrative(narrative_id: str) -> dict:
    for row in json.loads(NARRATIVES.read_text(encoding="utf-8"))["narratives"]:
        if row["narrative_id"] == narrative_id:
            return row
    raise ValueError(f"Narrative not found: {narrative_id}")


class ClaimState(TypedDict):
    narrative_id: str
    text: str
    claim_id: str
    intake_queue: str
    adjudication: str
    supervisor_notes: str
    event_trace: list[dict]


def supervisor(state: ClaimState) -> dict:
    narrative = load_narrative(state["narrative_id"])
    return {
        "text": narrative["text"],
        "supervisor_notes": "Route to intake, then adjudication.",
        "event_trace": [
            *state.get("event_trace", []),
            {"agent": "supervisor", "action": "assign_intake"},
        ],
    }


def intake_worker(state: ClaimState) -> dict:
    narrative = load_narrative(state["narrative_id"])
    return {
        "claim_id": narrative["claim_id"],
        "intake_queue": narrative["intake_queue"],
        "event_trace": [
            *state.get("event_trace", []),
            {
                "agent": "intake",
                "claim_id": narrative["claim_id"],
                "intake_queue": narrative["intake_queue"],
            },
        ],
    }


def adjudication_worker(state: ClaimState) -> dict:
    """Propose adjudication using intake outputs only.

    Guardrail: do not mutate claim_id or intake_queue owned by intake.
    """
    narrative = load_narrative(state["narrative_id"])
    return {
        "adjudication": narrative["adjudication_hint"],
        "event_trace": [
            *state.get("event_trace", []),
            {
                "agent": "adjudication",
                "adjudication": narrative["adjudication_hint"],
                "guardrail": "read_only_on_intake_fields",
                "seen_claim_id": state.get("claim_id", ""),
                "seen_intake_queue": state.get("intake_queue", ""),
            },
        ],
    }


def build_multi_agent():
    graph = StateGraph(ClaimState)
    graph.add_node("supervisor", supervisor)
    graph.add_node("intake", intake_worker)
    graph.add_node("adjudication", adjudication_worker)
    graph.add_edge(START, "supervisor")
    graph.add_edge("supervisor", "intake")
    graph.add_edge("intake", "adjudication")
    graph.add_edge("adjudication", END)
    return graph.compile()

    "claim_id": "",

Lesson A3 — Day 3 RAG: retrieve → grade → generate

Offline stub of the Day 3 PM graph. Real labs call Chroma / policy tools; here we grade with simple keyword evidence so the graph runs without an API key. 

# ASCII — Day 3 PM rag_graph
#
# START -> [retrieve] -> [grade] -> [generate] -> END
#
# retrieve: pull "documents" (here: policy snippets / narrative text)
# grade:    set sufficient_evidence bool
# generate: compose answer + citation ids


POLICY_SNIPPETS = {
    "auto": "Collision coverage applies when the insured vehicle is struck by another vehicle.",
    "water": "Sudden pipe bursts may be covered; flood requires a flood rider.",
    "cyber": "Ransomware restoration from backup may be covered under cyber endorsements.",
}


class RagState(TypedDict):
    question: str
    retrieval: dict
    answer: str
    citations: list[dict]
    sufficient_evidence: bool
    event_trace: list[dict]


def retrieve_node(state: RagState) -> dict:
    q = state["question"].lower()
    hits = []
    for doc_id, text in POLICY_SNIPPETS.items():
        if any(tok in q for tok in doc_id.split()) or any(
            w in text.lower() for w in q.split() if len(w) > 4
        ):
            hits.append({"doc_id": doc_id, "text": text})
    if not hits:
        # fallback: return all snippets so grade can still run
        hits = [{"doc_id": k, "text": v} for k, v in POLICY_SNIPPETS.items()]
    retrieval = {"steps": [{"retriever": "keyword", "hits": len(hits)}], "docs": hits}
    return {
        "retrieval": retrieval,
        "event_trace": [*state.get("event_trace", []), {"node": "retrieve", "hits": len(hits)}],
    }


def grade_node(state: RagState) -> dict:
    docs = state["retrieval"].get("docs", [])
    q = state["question"].lower()
    sufficient = any(
        any(w in d["text"].lower() for w in q.split() if len(w) > 4) for d in docs
    )
    return {
        "sufficient_evidence": sufficient,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "grade", "sufficient": sufficient, "n_docs": len(docs)},
        ],
    }


def generate_node(state: RagState) -> dict:
    docs = state["retrieval"].get("docs", [])
    if not state.get("sufficient_evidence"):
        answer = "Insufficient evidence in retrieved policies; escalate to human adjuster."
        citations: list[dict] = []
    else:
        top = docs[0]
        answer = f"Based on {top['doc_id']}: {top['text']}"
        citations = [{"doc_id": top["doc_id"], "quote": top["text"][:80]}]
    return {
        "answer": answer,
        "citations": citations,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "generate", "citations": len(citations)},
        ],
    }


def build_rag_graph():
    graph = StateGraph(RagState)
    graph.add_node("retrieve", retrieve_node)
    graph.add_node("grade", grade_node)
    graph.add_node("generate", generate_node)
    graph.add_edge(START, "retrieve")
    graph.add_edge("retrieve", "grade")
    graph.add_edge("grade", "generate")
    graph.add_edge("generate", END)
    return graph.compile()


rag = build_rag_graph()
