Beginner Client, collections, offline embeddings

1. PersistentClient vs ephemeral

June 2026 Python API (stable surface used in Week 2): 

import chromadb
from pathlib import Path

# In-memory only — gone when the process exits
# client = chromadb.Client()

PERSIST = Path("data/chroma_concept_building_lab").resolve()
PERSIST.mkdir(parents=True, exist_ok=True)

client = chromadb.PersistentClient(path=str(PERSIST))
print(client.list_collections())
print("Persist path:", PERSIST)

On open, Chroma loads existing collections from disk. On add / upsert, it writes under that path. Restart the kernel and the same PersistentClient(path=...) recovers the index.

3. Offline embedding functions (June 2026)

If you omit an embedding function, Chroma’s default is local Sentence Transformers all-MiniLM-L6-v2 (downloads once). For explicit, reproducible classroom code, attach SentenceTransformerEmbeddingFunction: 

from chromadb.utils.embedding_functions import (
    DefaultEmbeddingFunction,
    SentenceTransformerEmbeddingFunction,
)

# Option A — default (still local MiniLM)
# ef = DefaultEmbeddingFunction()

ef = SentenceTransformerEmbeddingFunction(model_name="all-MiniLM-L6-v2")

    metadata={"hnsw:space": "cosine"},

Same function for add and query The embedding function bound at collection create/get time encodes both documents and query_texts. Do not mix OpenAI embeddings in one process and MiniLM in another against the same persist dir — distances become meaningless.

Intermediate Chunk policies, upsert, query, filters

5. Chunk data/policies/*.md by ## headings

Mirror the classroom splitter: one chunk per ## section, stable ids, rich metadata for ACL and lineage. 

import re
from pathlib import Path

POLICIES = Path("data/policies")

def slug(value: str) -> str:
    cleaned = re.sub(r"[^a-zA-Z0-9]+", "-", value.lower()).strip("-")
    return cleaned or "section"

def split_markdown(path: Path) -> list[dict]:
    text = path.read_text(encoding="utf-8")
    policy_id = path.stem.split("_")[0]  # POL-AUTO-001_coverage -> POL-AUTO-001
    chunks, heading, buffer = [], "intro", []
    for line in text.splitlines():
        if line.startswith("## "):
            if buffer:
                body = "\n".join(buffer).strip()
                if body:
                    chunks.append({
                        "chunk_id": f"{path.stem}#{slug(heading)}",
                        "policy_id": policy_id,
                        "source_file": path.name,
                        "heading": heading,
                        "text": body,
                        "lane": "insurance",
                        "version": "v1",
                    })
                buffer = []
            heading = line[3:].strip()
        else:
            buffer.append(line)
    body = "\n".join(buffer).strip()
    if body:
        chunks.append({
            "chunk_id": f"{path.stem}#{slug(heading)}",
            "policy_id": policy_id,
            "source_file": path.name,
            "heading": heading,
            "text": body,
            "lane": "insurance",
            "version": "v1",
        })
    return chunks

all_chunks = []
for path in sorted(POLICIES.glob("*.md")):
    all_chunks.extend(split_markdown(path))

print(f"{len(list(POLICIES.glob('*.md')))} files → {len(all_chunks)} chunks")
for c in all_chunks[:3]:
    print(c["chunk_id"], "|", c["heading"], "|", c["text"][:60].replace("\n", " "), "...")

7. Semantic query with query_texts

def show_hits(results, title="hits"):
    print(title)
    for doc, meta, dist, cid in zip(
        results["documents"][0],
        results["metadatas"][0],
        results["distances"][0],
        results["ids"][0],
    ):
        print(f"  dist={dist:.4f} | {cid}")
        print(f"    {meta['policy_id']} · {meta['heading']}")
        print(f"    {doc[:120].replace(chr(10), ' ')}...")

results = collection.query(
    query_texts=["Is rental car reimbursement included after theft?"],
    n_results=3,
    include=["documents", "metadatas", "distances"],
)
show_hits(results, "Top-3 semantic")

With hnsw:space: cosine, smaller distance ≈ closer. Always keep ids in the result for citations in grounded answers.

8. Metadata where filters

Filters restrict the candidate set before / as part of ANN search — critical for permission-aware retrieval (Day 3 AM ACL demos). 

# Equality
auto_only = collection.query(
    query_texts=["deductible for comprehensive claims"],
    n_results=3,
    where={"policy_id": "POL-AUTO-001"},
    include=["documents", "metadatas", "distances"],
)
show_hits(auto_only, "where policy_id = POL-AUTO-001")

# $and / $or / $ne / $in (Chroma where operators)
home_water = collection.query(
    query_texts=["water damage from burst pipe"],
    n_results=3,
    where={
        "$and": [
            {"policy_id": {"$in": ["POL-HOME-010"]}},
            {"lane": "insurance"},
        ]
    },
    include=["documents", "metadatas", "distances"],
)
show_hits(home_water, "home + insurance lane")

Document content filters use where_document (e.g. {"$contains": "deductible"}) — use sparingly; prefer structured metadata for production ACL.

Advanced Rebuild, governance, agentic RAG hookup

10. Idempotent rebuild helper

Classroom pattern: env flag REBUILD_INDEX. For ConceptBuilding labs, encapsulate chunk → upsert so you can wipe and refresh after editing markdown. 

import os

def rebuild_policy_lab(client, persist_path: Path, collection_name: str = "policy_chunks_lab"):
    """Chunk data/policies and upsert into a persistent collection."""
    from chromadb.utils.embedding_functions import SentenceTransformerEmbeddingFunction

    chunks = []
    for path in sorted(Path("data/policies").glob("*.md")):
        chunks.extend(split_markdown(path))
    if not chunks:
        raise SystemExit("No policy markdown under data/policies")

    ef = SentenceTransformerEmbeddingFunction(model_name="all-MiniLM-L6-v2")
    col = client.get_or_create_collection(name=collection_name, embedding_function=ef)
    existing = col.get()
    if existing.get("ids"):
        col.delete(ids=existing["ids"])
    col.upsert(
        ids=[c["chunk_id"] for c in chunks],
        documents=[c["text"] for c in chunks],
        metadatas=[{
            "policy_id": c["policy_id"],
            "source_file": c["source_file"],
            "heading": c["heading"],
            "lane": c["lane"],
            "version": c["version"],
        } for c in chunks],
    )
    print(f"Rebuilt {collection_name} at {persist_path} ({len(chunks)} chunks)")
    return col

if os.getenv("REBUILD_INDEX", "true").lower() == "true":
    collection = rebuild_policy_lab(client, PERSIST)

11. Retrieve tool shape for agentic RAG

Single-pass RAG: retrieve once → stuff context → answer. Agentic RAG: the agent (or LangGraph node) may re-query with a refined question, tighter where, or higher n_results after a grader says evidence is weak. Chroma is the tool backend, not the agent. 

from typing import Any

def retrieve_policy(
    question: str,
    *,
    n_results: int = 4,
    policy_id: str | None = None,
    collection=collection,
) -> list[dict[str, Any]]:
    """Operator-facing retrieve tool — returns citation-ready dicts."""
    where = {"policy_id": policy_id} if policy_id else None
    raw = collection.query(
        query_texts=[question],
        n_results=n_results,
        where=where,
        include=["documents", "metadatas", "distances"],
    )
    hits = []
    for i in range(len(raw["ids"][0])):
        hits.append({
            "chunk_id": raw["ids"][0][i],
            "text": raw["documents"][0][i],
            "metadata": raw["metadatas"][0][i],
            "distance": raw["distances"][0][i],
        })
    return hits

# --- agentic loop sketch (no LLM required for this module) ---
question = "Does flood damage require a rider?"
hits = retrieve_policy(question, n_results=3)
enough = any("flood" in h["text"].lower() or "flood" in h["metadata"]["heading"].lower() for h in hits)
if not enough:
    hits = retrieve_policy(
        "flood rider rising water exclusion homeowners",
        n_results=5,
        policy_id="POL-HOME-010",
    )
for h in hits:
    print(h["chunk_id"], h["distance"], h["metadata"]["heading"])

# Example: tag sensitivity when upserting (lab only)
# metadatas include e.g. {"acl_role_min": "teller"} or {"audience": "internal"}

# At query time — never trust the model to self-filter
role = "teller"
allowed = {"teller", "adjuster", "underwriter"}  # demo set
assert role in allowed
results = collection.query(
    query_texts=["customer deductible"],
    n_results=3,
    where={"lane": "insurance"},  # stand-in; real ACL uses role-tagged metadata
)
Never invent records Vector hits are still facts only if the source markdown said so. Grounded answers must cite chunk_id / source_file. If retrieval is empty, the agent should say evidence is missing — not hallucinate a clause.
