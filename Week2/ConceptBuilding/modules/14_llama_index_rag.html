Exercise B3 — Split markdown on ## headings

Mirror the shipped chunk contract: one chunk per section with stable citation identity.

import re

def slug(value: str) -> str:
    cleaned = re.sub(r"[^a-zA-Z0-9]+", "-", value.lower()).strip("-")
    return cleaned or "section"

def split_markdown(path: Path) -> list[dict]:
    policy_id = path.stem.split("_")[0]
    chunks, heading, buffer = [], "intro", []
    for line in path.read_text(encoding="utf-8").splitlines():
        if line.startswith("## "):
            if buffer:
                chunks.append({
                    "chunk_id": f"{path.stem}#{slug(heading)}",
                    "policy_id": policy_id,
                    "source_file": path.name,
                    "heading": heading,
                    "text": "\n".join(buffer).strip(),
                })
                buffer = []
            heading = line[3:].strip()
        else:
            buffer.append(line)
    if buffer:
        chunks.append({
            "chunk_id": f"{path.stem}#{slug(heading)}",
            "policy_id": policy_id,
            "source_file": path.name,
            "heading": heading,
            "text": "\n".join(buffer).strip(),
        })
    return chunks

Exercise B5 — Tokenize + keyword score

Same idea as Week2 data/rag/retrieval_utils.py — no BM25 library needed.

import math
from collections import Counter

def tokenize(text: str) -> list[str]:
    return re.findall(r"[a-z0-9]+", text.lower())

def keyword_score(query: str, text: str) -> float:
    q, d = tokenize(query), tokenize(text)
    if not q or not d:
        return 0.0
    counts = Counter(d)
    score = 0.0
    for token in set(q):
        tf = counts.get(token, 0)
        if tf:
            score += 1.0 + math.log(1.0 + tf)
    return score

q = "rental car after theft out of state"
for c in corpus:
    c["score"] = keyword_score(q, c["text"] + " " + c["heading"])
ranked = sorted(corpus, key=lambda x: x["score"], reverse=True)[:5]
for c in ranked:
    print(f"{c['score']:.2f}  {c['chunk_id']}")
Exercise B6 — Single-pass top-k helper

One retrieval round only. This is the baseline Day 3 demo compares against agentic.

def top_k_chunks(query: str, chunks: list[dict], k: int = 3) -> list[dict]:
    scored = [
        {**c, "score": keyword_score(query, c["text"] + " " + c["heading"])}
        for c in chunks
    ]
    return sorted(scored, key=lambda x: x["score"], reverse=True)[:k]

def single_pass_retrieve(query: str, chunks: list[dict], k: int = 3) -> dict:
    hits = top_k_chunks(query, chunks, k=k)
    return {
        "mode": "single_pass",
        "query": query,
        "steps": [{"round": 0, "query": query, "hits": [h["chunk_id"] for h in hits]}],
        "chunks": hits,
    }

Exercise B8 — Grade evidence (offline)

Return sufficient and missing — the heart of agentic RAG.

def grade_evidence(question_id: str | None, chunk_ids: list[str]) -> dict:
    if not question_id:
        return {"sufficient": True, "missing": []}
    golden = by_id.get(question_id)
    if golden is None:
        return {"sufficient": True, "missing": []}
    required = set(golden["required_chunk_ids"])
    missing = sorted(required - set(chunk_ids))
    return {"sufficient": not missing, "missing": missing, "question_id": question_id}

Exercise I6 — Refine query helper

When grade fails, append heading tokens from current hits (same idea as retrieval_utils).

def refine_query(base_query: str, hits: list[dict]) -> str:
    terms = []
    for hit in hits:
        terms.extend(tokenize(hit.get("heading", "")))
    extra = " ".join(sorted(set(terms))[:4])
    return f"{base_query} {extra}".strip()

Exercise I7 — Agentic retrieve loop (keyword, offline)

Max 3 rounds. Stop when grade["sufficient"]. Prefer this in class without API spend.

def dedupe_chunks(chunks: list[dict]) -> list[dict]:
    seen, out = set(), []
    for c in chunks:
        if c["chunk_id"] in seen:
            continue
        seen.add(c["chunk_id"])
        out.append(c)
    return out

def agentic_retrieve(query, chunks, question_id=None, max_rounds=3, k=2):
    steps, collected = [], []
    current, grade = query, {"sufficient": False, "missing": []}
    for r in range(max_rounds):
        hits = top_k_chunks(current, chunks, k=k)
        collected = dedupe_chunks(collected + hits)
        ids = [c["chunk_id"] for c in collected]
        grade = grade_evidence(question_id, ids)
        steps.append({
            "round": r,
            "query": current,
            "hits": [h["chunk_id"] for h in hits],
            "sufficient": grade["sufficient"],
            "missing": grade.get("missing", []),
        })
        if grade["sufficient"]:
            break
        current = refine_query(query, hits)
    return {
        "mode": "agentic_rag",
        "query": query,
        "steps": steps,
        "chunks": collected,
        "grade": grade,
    }

agentic = agentic_retrieve(G001, corpus, question_id="G-001", max_rounds=3, k=2)
print("rounds:", len(agentic["steps"]), "sufficient:", agentic["grade"]["sufficient"])
for s in agentic["steps"]:

Exercise I8 — Dense agentic round (retriever.retrieve)

Same loop, but each round calls LlamaIndex retriever instead of keyword top-k.

def agentic_dense(query, retriever, question_id, max_rounds=3):
    steps, collected_ids, collected_meta = [], [], []
    current = query
    grade = {"sufficient": False, "missing": []}
    for r in range(max_rounds):
        nodes = retriever.retrieve(current)
        hits = []
        for n in nodes:
            cid = n.node.metadata.get("chunk_id") or n.node.node_id
            hits.append({
                "chunk_id": cid,
                "heading": n.node.metadata.get("heading", ""),
                "text": n.text,
                "score": n.score,
            })
            if cid not in collected_ids:
                collected_ids.append(cid)
                collected_meta.append(hits[-1])
        grade = grade_evidence(question_id, collected_ids)
        steps.append({
            "round": r,
            "query": current,
            "hits": [h["chunk_id"] for h in hits],
            "missing": grade.get("missing", []),
            "sufficient": grade["sufficient"],
        })
        if grade["sufficient"]:
            break
        current = refine_query(query, hits)
    return {"steps": steps, "chunks": collected_meta, "grade": grade}

dense_agentic = agentic_dense(G001, retriever, "G-001")
print(dense_agentic["grade"])
for s in dense_agentic["steps"]:
    print(s["round"], s["hits"], "missing=", s["missing"])

Exercise A3 — Offline compose_answer (no LLM)

Template answer from headings + sufficient flag. Stretch later swaps in an LLM.

def compose_answer(question: str, retrieval: dict) -> GroundedAnswer:
    chunks = retrieval.get("chunks", [])
    grade = retrieval.get("grade", {})
    cites = make_citations(chunks)
    if not grade.get("sufficient", False):
        answer = (
            "Insufficient evidence for a full answer. Missing: "
            + ", ".join(grade.get("missing", []))
        )
    else:
        headings = "; ".join(c.heading for c in cites)
        answer = (
            f"Based on policy sections [{headings}], the combined clauses address: {question}"
        )
    return GroundedAnswer(
        lane="insurance",
        question=question,
        answer=answer,
        citations=cites,
        retrieval_steps=retrieval.get("steps", []),
        sufficient_evidence=bool(grade.get("sufficient")),
    )

ga = compose_answer(G001, agentic)

Exercise A5 — LangGraph-shaped nodes (offline)

Same topology as the W2D3 AM topic 01 grounded loop: retrieve → grade → generate.

from typing import TypedDict

class RagState(TypedDict):
    question: str
    question_id: str | None
    retrieval: dict
    answer: str
    citations: list[dict]
    sufficient_evidence: bool
    event_trace: list[dict]

def retrieve_node(state: RagState) -> dict:
    retrieval = agentic_retrieve(
        state["question"], corpus, question_id=state.get("question_id"), max_rounds=3, k=2
    )
    return {
        "retrieval": retrieval,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "retrieve", "rounds": len(retrieval["steps"])},
        ],
    }

def grade_node(state: RagState) -> dict:
    grade = state["retrieval"].get("grade", {})
    return {
        "sufficient_evidence": bool(grade.get("sufficient")),
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "grade", "sufficient": grade.get("sufficient"), "missing": grade.get("missing", [])},
        ],
    }

def generate_node(state: RagState) -> dict:
    ga = compose_answer(state["question"], state["retrieval"])
    return {
        "answer": ga.answer,
        "citations": [c.model_dump() for c in ga.citations],
        "sufficient_evidence": ga.sufficient_evidence,
        "event_trace": [
            *state.get("event_trace", []),
            {"node": "generate", "citations": len(ga.citations)},
        ],
    }

state: RagState = {
    "question": G001,
    "question_id": "G-001",
    "retrieval": {},
    "answer": "",
    "citations": [],
    "sufficient_evidence": False,
    "event_trace": [],
}
state = {**state, **retrieve_node(state)}
state = {**state, **grade_node(state)}
state = {**state, **generate_node(state)}

Exercise A6 — Conditional refine edge

If grade fails after retrieve, route to refine; else generate. Sketch routing function.

def route_after_grade(state: RagState) -> str:
    if state.get("sufficient_evidence"):
        return "generate"
    rounds = len(state.get("retrieval", {}).get("steps", []))
    if rounds >= 3:
        return "generate"  # stop and admit insufficient
    return "refine"

Exercise A7 — Batch goldens eval (offline)

Score how many questions become sufficient under agentic vs single-pass.

def eval_mode(mode: str) -> dict:
    ok = 0
    rows = []
    for g in goldens:
        qid, q = g["question_id"], g["question"]
        if mode == "single":
            result = single_pass_retrieve(q, corpus, k=3)
            ids = result["steps"][0]["hits"]
            grade = grade_evidence(qid, ids)
        else:
            result = agentic_retrieve(q, corpus, question_id=qid, max_rounds=3, k=2)
            grade = result["grade"]
        ok += int(grade["sufficient"])
        rows.append((qid, grade["sufficient"], grade.get("missing", [])))
    return {"mode": mode, "n_ok": ok, "n": len(goldens), "rows": rows}

Exercise A9 — Hybrid: keyword OR dense union before grade

When dense alone misses a clause, union keyword hits for the same query.

def hybrid_hits(query: str, k: int = 3) -> list[dict]:
    kw = top_k_chunks(query, corpus, k=k)
    dense = retriever.retrieve(query)
    by_id_map = {c["chunk_id"]: c for c in corpus}
    merged = {c["chunk_id"]: c for c in kw}
    for n in dense:
        cid = n.node.metadata.get("chunk_id")
        if cid and cid in by_id_map:
            merged[cid] = by_id_map[cid]
    return list(merged.values())

hyb = hybrid_hits(G001, k=3)
print([c["chunk_id"] for c in hyb])
print(grade_evidence("G-001", [c["chunk_id"] for c in hyb]))
